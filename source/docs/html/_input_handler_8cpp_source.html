41bool InputHandler::validateDate(const std::string &date)
42{
43 // Check format DD-MM-YYYY
44 const std::regex pattern(R"(\d{2}-\d{2}-\d{4})");
45 if (!std::regex_match(date, pattern))
46 {
47 return false;
48 }
49
50 // Extract day, month, and year
51 int day, month, year;
52 sscanf(date.c_str(), "%d-%d-%d", &day, &month, &year);
53
54 // Check valid month
55 if (month < 1 || month > 12)
56 {
57 return false;
58 }
59
60 // Check valid day based on month
61 const int daysInMonth[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
62
63 // Adjust for leap year
64 int maxDays = daysInMonth[month];
65 if (month == 2 && ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)))
66 {
67 maxDays = 29;
68 }
69
70 if (day < 1 || day > maxDays)
71 {
72 return false;
73 }
74
75 return true;
76}
