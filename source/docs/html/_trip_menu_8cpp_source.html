179bool TripMenu::isEndDateValid(const std::string &startDate, const std::string &endDate)
180{
181 // Format: DD-MM-YYYY
182 // Extract individual components
183 int startDay, startMonth, startYear;
184 int endDay, endMonth, endYear;
185
186 sscanf(startDate.c_str(), "%d-%d-%d", &startDay, &startMonth, &startYear);
187 sscanf(endDate.c_str(), "%d-%d-%d", &endDay, &endMonth, &endYear);
188
189 // Compare years first
190 if (endYear > startYear) {
191 return true;
192 }
193 else if (endYear < startYear) {
194 return false;
195 }
196
197 // Same year, compare months
198 if (endMonth > startMonth) {
199 return true;
200 }
201 else if (endMonth < startMonth) {
202 return false;
203 }
204
205 // Same year and month, compare days
206 return endDay > startDay;
207}
