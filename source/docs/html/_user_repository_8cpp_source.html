235User UserRepository::parseUserFromCsv(const std::string &line)
236{
237 // Skip empty lines
238 if (line.empty())
239 {
240 return User(0, "", "", "", "");
241 }
242
243 // Skip header line
244 if (line.substr(0, 2) == "id")
245 {
246 return User(0, "", "", "", "");
247 }
248
249 std::stringstream ss(line);
250 std::string idStr, name, address, phone, email;
251
252 // Parse CSV format: id,name,address,phone,email
253 std::getline(ss, idStr, ',');
254 std::getline(ss, name, ',');
255 std::getline(ss, address, ',');
256 std::getline(ss, phone, ',');
257 std::getline(ss, email);
258
259 try
260 {
261 int id = std::stoi(idStr);
262 return User(id, name, address, phone, email);
263 }
264 catch (const std::exception &e)

266 std::cerr << "Error parsing user ID from CSV: " << e.what() << " (line: " << line << ")" << std::endl;
